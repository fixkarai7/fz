PRACTICAL NO. 5  -  MongoDB : MapReduce - Average Age Calculation

// Insert data
db.person.insertMany([
{ "_id": 1, "name": "Alice", "age": 25, "city": "New York" },
{ "_id": 2, "name": "Bob", "age": 30, "city": "San Francisco" },
{ "_id": 3, "name": "Charlie", "age": 35, "city": "Los Angeles" },
{ "_id": 4, "name": "David", "age": 40, "city": "Chicago" },
{ "_id": 5, "name": "Eve", "age": 28, "city": "Miami" }
])

// Map function
var mapFunction = function() { emit("average_age", {
total: this.age, count: 1
});
};

// Reduce function
var reduceFunction = function(key, values) { var result = {
total: 0,
count: 0
};
values.forEach(function(value) { result.total += value.total; result.count += value.count;
});
return result;
};

// Finalize function
var finalizeFunction = function(key, value) { return value.total / value.count;
};

// Execute
db.person.mapReduce( mapFunction, reduceFunction,
{
out: "average_age", finalize: finalizeFunction
}
)

// Display result
db.average_age.find()

# Q3. Reverse Order Retrieval - descending order of scores
ZREVRANGE leaderboard 0 -1 WITHSCORES

# Q4. Adding Members - add David with score 1800
ZADD leaderboard 1800 David

# Q5. Ranking Position of Alice
# Alice's rank in ascending score order
ZRANK leaderboard Alice
# Rank from highest score to lowest
ZREVRANK leaderboard Alice

# Q6. Removing Members - remove Charlie
ZREM leaderboard Charlie

# Q7. Range Removal - remove players with score below 1500
ZREMRANGEBYSCORE leaderboard -inf (1500

# Q8. Count members with scores between 1500 and 2500
ZCOUNT leaderboard 1500 2500

# Q9. Update Bob's score - increase by 200
ZINCRBY leaderboard 200 Bob

# Q10. Combining Sorted Sets
# Step 1 - Create the morning leaderboard and evening leaderboard
ZADD morning_leaderboard 100 Alice 200 Bob 150 Charlie
ZADD evening_leaderboard 200 Alice 100 Bob 250 David
# Step 2 - Combine them
ZUNIONSTORE daily_leaderboard 2 morning_leaderboard evening_leaderboard AGGREGATE SUM

PRACTICAL NO. 10  -  Python + Redis : Task Queue using List



----- CODE -----
# 1. Add five tasks
tasks = [
"Task 1: Check emails", "Task 2: Process data", "Task 3: Generate report", "Task 4: Backup database", "Task 5: Send report"
]
for task in tasks:
    r.rpush("task_queue", task)
print("Five tasks added successfully")
print(r.lrange("task_queue", 0, -1))

# 2. Remove tasks from the left and process them
while r.llen("task_queue") > 0:
    task = r.lpop("task_queue")
    print("Processed:", task)

# 3. Display remaining tasks
print("Remaining tasks:", r.lrange("task_queue", 0, -1))
